  no/.style={draw=red!55!black, fill=red!5, rounded corners=2pt, align=left, text width=3.7cm, minimum height=13mm, inner sep=2pt},

\node[q, fill=black!4] (c) {\textbf{Candidate input}};
\node[q, below=of c] (q1) {1. Known on release day? (no label information from the song's own period or later)};
\node[q, below=of q1] (q2) {2. Gain under CV,\\corrected $p<0.05$?};
\node[q, below=of q2] (q3) {3. Survives the controls?\\year-, genre-matched; unseen artists};
\node[q, below=of q3] (q4) {4. Survives Bonferroni\\($p<0.05/400$)?};
\node[yes, below=of q4] (k) {\textbf{kept}: artist history; decade\\(Hit Predictor); lyrics (ISMIR)};
\node[no, right=3mm of q1] (n1) {\textbf{out} -- leak: genre tags (assigned afterwards); artist history on SpotGenTrack (upper bound)};

\draw[->] (c) -- (q1);
\foreach \a/\b in {q1/q2, q2/q3, q3/q4, q4/k} \draw[->] (\a) -- node[left, font=\tiny] {yes} (\b);
\foreach \a/\b in {q1/n1, q2/n2, q3/n3, q4/n4} \draw[->] (\a) -- node[above, font=\tiny] {no} (\b);
\end{tikzpicture}
\caption{How every candidate input was kept or rejected. Each input is tested in this order; the boxes on the right
list the inputs that stopped at each question. An input \emph{fails} question 3 when, under at least one of the
controls we ran on it (each applied separately, or combined where stated), its gain is no longer significant after
Bonferroni. The after-release trajectory model answers a
different question and is outside this flowchart.}
